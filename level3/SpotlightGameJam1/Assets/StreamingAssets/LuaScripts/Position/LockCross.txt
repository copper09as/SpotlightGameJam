function LockCross(entity)
    local path = "Character/CharacterCrossMap"
    local list = entity.scriptData.UpdatePath
    local index = -1

    for i = 0, list.Count - 1 do
        if list[i] == path then
            index = i
            break
        end
    end

    local rb = entity.rb

    if index ~= -1 then
        list:RemoveAt(index)

        if entity.sr ~= nil then
            entity.sr.color = CS.UnityEngine.Color.red
        end

        if rb ~= nil then
            rb.mass = 10000
            rb.velocity = CS.UnityEngine.Vector2(0, 0)
            rb.angularVelocity = 0
        end
    else
        list:Add(path)

        if entity.sr ~= nil then
            entity.sr.color = CS.UnityEngine.Color.white
        end

        if rb ~= nil then
            rb.mass = 1
        end
    end
end
