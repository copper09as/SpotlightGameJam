function CancleStick(entity)
    if entity == nil then return end

    local data = entity.dataTable
    local EM = CS.EntityUIManager.Instance

    if data.hitCharacters == nil or EM == nil then return end

    EM:CallNextFrame(function()
        for i = 1, #data.hitCharacters do
            local character = data.hitCharacters[i]
            if character and character.transform then
                character.transform:SetParent(nil, true)
                character.gameObject:SetActive(true)
            end
        end
        data.hitCharacters = {}
    end)
end
