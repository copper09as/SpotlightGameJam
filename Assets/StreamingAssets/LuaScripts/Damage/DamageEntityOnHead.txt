local GC = CS.GameController


function DamageEntityOnHead(entity, otherEntity, x, y)
    if y ~= nil and y > 0.5 then
        if otherEntity ~= nil then
            otherEntity:Dead(entity)
        end
		CS.UnityEngine.Debug.Log(otherEntity.dataTable.isEnemy)
		if otherEntity.dataTable.isEnemy then
			entity:PlayDamageObj()
		end
    end
end
