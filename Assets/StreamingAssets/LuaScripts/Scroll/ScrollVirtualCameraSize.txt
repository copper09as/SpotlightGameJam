local GC = CS.GameController

local maxSize = 4.71
local minSize = 1

function ScrollVirtualCameraSize(entity)
	if entity == nil then
		return
	end
    local data = entity.dataTable
	if data == nil then
		return
	end
    local t = data.scrollProgress
	
    t = math.clamp(t, 0, 1)

    local targetSize = maxSize + (minSize - maxSize) * t
	CS.CameraSwitch.Instance:SetTheCinameSize(targetSize)
end

function math.clamp(val, min, max)
    if val < min then return min end
    if val > max then return max end
    return val
end
