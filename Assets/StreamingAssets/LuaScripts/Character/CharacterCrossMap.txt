local GC = CS.GameController

function CharacterCrossMap(entity)
    local cam = CS.UnityEngine.Camera.main
    local pos = entity.transform.position
    local data = entity.dataTable

    -- 获取屏幕坐标
    local screenPos = cam:WorldToScreenPoint(pos)
    local screenWidth = CS.UnityEngine.Screen.width
    local screenHeight = CS.UnityEngine.Screen.height
    local offset = 1 -- 像素偏移，防止重复传送

    -- 初始化穿越记录
    if data.lastCrossDirection == nil then
        data.lastCrossDirection = { x = 0, y = 0 }
    end

    local crossed = false
    local dirX, dirY = 0, 0  -- -1 左/下，1 右/上

    -- ✅ 超出左右边界
    if screenPos.x < 0 then
        dirX = -1
        crossed = true
    elseif screenPos.x > screenWidth then
        dirX = 1
        crossed = true
    end

    -- ✅ 超出上下边界
    if screenPos.y < 0 then
        dirY = -1
        crossed = true
    elseif screenPos.y > screenHeight then
        dirY = 1
        crossed = true
    end

    -- 处理水平穿越
    if dirX ~= 0 then
        if dirX == data.lastCrossDirection.x then
            -- 重复穿越，轻微调整
            local adjust = (dirX == -1) and 10 or -10
            screenPos.x = screenPos.x + adjust
        else
            -- 正常穿越
            if dirX == -1 then
                screenPos.x = screenWidth - offset
            else
                screenPos.x = 0 + offset
            end
            data.lastCrossDirection.x = dirX
        end
    else
        data.lastCrossDirection.x = 0
    end

    -- 处理垂直穿越
    if dirY ~= 0 then
        if dirY == data.lastCrossDirection.y then
            -- 重复穿越，轻微调整
            local adjust = (dirY == -1) and 10 or -10
            screenPos.y = screenPos.y + adjust
        else
            -- 正常穿越
            if dirY == -1 then
                screenPos.y = screenHeight - offset
            else
                screenPos.y = 0 + offset
            end
            data.lastCrossDirection.y = dirY
        end
    else
        data.lastCrossDirection.y = 0
    end

    -- 将屏幕坐标转换回世界坐标
    local worldPos = cam:ScreenToWorldPoint(screenPos)
    entity.transform.position = CS.UnityEngine.Vector3(worldPos.x, worldPos.y, pos.z)
end
