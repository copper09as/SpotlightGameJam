local function DragonFly(entity)--飞回原来的位置
    CS.UnityEngine.Debug.Log("恶龙飞到随机的点位")--点位提前放置
    local data = entity.dataTable
    entity.transform:DOMove( CS.BossStopPoint.Instance:GetPoint(),data.flyTiming):SetEase(CS.DG.Tweening.Ease.InOutCubic)
          :OnComplete(function ()
        data.state = "idle"
        CS.UnityEngine.Debug.Log("飞回完成")
        data.attackTimer = data.attackCooldown --重置攻击冷却时间计时器
    end)
end
local function GetRelevantValue(a,b)--最后一关返回a ，其余关返回b
    if CS.BattleConfig.Instance.levelId == 14 then
        return a
    end
    return b
end
function DragonInit(entity)
    local data = entity.dataTable or {}

    -- 基础属性
    data.ccbTime = GetRelevantValue(8,1000)                --被踩背次数

    data.speed = 3.5                  -- 普通移动速度
    data.gravity = 0
    data.jumpForce = math.sqrt(2 * data.gravity * 3) -- 默认跳跃高度 3
    data.isJumping = false
    data.isFalling = false

    data.beforeTime = 1              --攻击前摇时间
    data.afterTime = 2               --攻击后摇时间
    data.beforeTimer = 1             --攻击前摇计时器
    data.afterTimer = 2              --攻击后摇计时器
    data.attackCooldown = 2          -- 攻击冷却时间
    data.attackTimer = 0
    data.attackTime = 2               --龙攻击到角色的时间
    data.attackPosition = nil         --攻击前往的位置
    data.sprintSpeed = 10             -- 冲刺速度
    data.flyTiming = 4                -- 飞行时间

    data.target = CS.UnityEngine.GameObject.Find("Character")       -- 追踪目标
    data.originPosition = entity.transform.position   --龙在场景初始位置
    data.canAttack = true
    data.canSprint = true
    data.invincibleTime = 3           -- 无敌时间
    data.isInvincible = false         -- 是否无敌
    data.flyTime = 0                  -- 飞行计时
    data.flyDuration = 3              -- 飞行持续时间
    data.DMRadius = 10                -- 最远攻击距离

	data.isEnemy= true
    data.isDragon = true
    data.colliders = CS.Tool.GetCollider2Ds(entity.gameObject)
    data.state = "flyBack"               -- Boss 状态
    --idle（fly）/attack/attackBefore/attackAfter/
    DragonFly(entity)                 --boss从场外飞入场内随机一个点位

end
