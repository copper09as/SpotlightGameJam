function DragonInit(entity)
    local data = entity.dataTable or {}

    -- 基础属性
    data.speed = 3.5                  -- 普通移动速度
    data.gravity = 9.8
    data.jumpForce = math.sqrt(2 * data.gravity * 3) -- 默认跳跃高度 3
    data.isJumping = false
    data.isFalling = false
    data.attackCooldown = 2           -- 攻击冷却时间
    data.attackTimer = 0
    data.attackPosition = nil         --攻击前往的位置
    --data.sprintCooldown = 5           -- 冲刺冷却
    --data.sprintTimer = 0
    data.sprintSpeed = 10             -- 冲刺速度
    data.flyHeight = 4                -- 飞行高度
    data.state = "idle"               -- Boss 状态 idle（fly）/attack（sprint）
    data.target = CS.UnityEngine.GameObject.Find("Character")       -- 追踪目标
    data.canAttack = true
    data.canSprint = true
    data.flyTime = 0                  -- 飞行计时
    data.flyDuration = 3               -- 飞行持续时间
	data.positionLerpSpeed = 10
	
end
