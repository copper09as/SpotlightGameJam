local CE = CS.XLuaCustomExport

local function DragonWakeFly(entity)
    if entity.col then
        entity.col.enabled = false
    end
    --if entity.sr then
    --    local color = entity.sr.color
    --    color.a = 0.5
    --    entity.sr.color = color
    --end
    if entity.rb then
        entity.rb.velocity = CS.UnityEngine.Vector2(10, 10)
        entity.rb.angularVelocity = 0
    end
    if entity.animator then
        entity.animator:ResetTrigger("StopAttack")
        entity.animator:SetBool("CanAttack",false)
        entity.animator:Play("Fly")
    end
    entity.transform.localScale =  CS.UnityEngine.Vector3(-1, 1, 1)

    entity.entityManager:DestroyEntityDelay(entity,8)

    --放胜利的逻辑

end
local function DragonInvincible(entity) --龙进入无敌状态
    local data = entity.dataTable
    --状态处理
    data.isInvincible = true
    --entity.animator:SetTrigger("BeAttack")
    CS.UnityEngine.Debug.Log("被攻击一次")
    CS.UnityEngine.Debug.Log("调用飞行方法")
    data.afterTimer = data.afterTime--重置后摇计时器

    data.state = "flyBack"
    entity.animator:SetBool("CanAttack",false)
    entity.animator:SetTrigger("Fly")
    entity.animator:ResetTrigger("StopAttack")--保险起见清楚没有被触发的trigger
    DragonFly(entity)

    local colorA = entity.sr.color
    local colorB = entity.sr.color
    colorA.b = 0.5
    colorA.g = 0.5
    for i = 0,data.colliders.Length - 1 do
        local collider = data.colliders[i]
        collider.enabled = false --关闭碰撞体
    end
    CE.ChangeColor(entity.sr,colorA,0.2,
            function()
                colorB.b = 0.5
                CE.ChangeColor(entity.sr,colorB,data.invincibleTime,
                        function()
                            colorB.b = 1;
                            entity.sr.color = colorB
                            --此时无敌时间结束
                            data.isInvincible = false
                            for i = 0,data.colliders.Length - 1 do
                                local collider = data.colliders[i]
                                collider.enabled = true--关闭碰撞体
                            end
                        end)
            end)

end
function DragonBeAttacked(entity,otherEntity)
    if otherEntity~= nil and otherEntity:CompareTag("Enemy") then
        print("被怪物碰撞不死亡")
        return
    end
    local data = entity.dataTable

	if not data.isInvincible then
		data.ccbTime = data.ccbTime - 1

		local liveFireList = entity:GetEntities("Live")


		if liveFireList ~= nil and liveFireList.Count > 0 then
			local index = liveFireList.Count - data.ccbTime

			if index >= 1 and index <= liveFireList.Count then
				local fireEntity = liveFireList[index-1]
				if fireEntity ~= nil and fireEntity.dataTable ~= nil then
					fireEntity.dataTable.live = false
				end
			end
		end

        if data.ccbTime <= 0 then
            DragonWakeFly(entity) -- 龙飞走
        else
            DragonInvincible(entity)
        end
	end


    if otherEntity ~= nil and otherEntity.rb ~= nil then
        local velocity = otherEntity.rb.velocity
        velocity.y = 10
        otherEntity.rb.velocity = velocity
    end

end
