local Mathf = CS.UnityEngine.Mathf
local Vector3 = CS.UnityEngine.Vector3
local AM = CS.AudioManager.Instance
local function SetDirection(position,Dragon) --设置朝向

    local Direction = position - Dragon.transform.position

    if Direction.x <=0 then
        Dragon.transform.localScale = Vector3(1, 1, 1)
    else
        Dragon.transform.localScale = Vector3(-1, 1, 1)
    end
end

function DragonUpdate(entity, dt)
    local data = entity.dataTable
    local target = data.target
    if target == nil then return end

    -- 更新冷却计时
    data.attackTimer = math.max(0, data.attackTimer - dt)
    CS.UnityEngine.Debug.Log("龙的状态"..data.state)
    -- 根据状态执行行为
    if data.state == "idle" then --待机不执行动作
        DragonIdle(entity, dt)
    elseif data.state == "attackBefore" then
        DragonAttackBefore(entity, dt)
    elseif data.state == "attackAfter" then
        DragonAttackAfter(entity, dt,1)
    --elseif data.state == "flyBack" and "find"then
    --
    end
end


function DragonIdle(entity) --什么都不做
    local data = entity.dataTable

    local Direction = data.target.transform.position - entity.transform.position

    if Direction.x > 0 then
        data.Dragon.transform.localScale = Vector3(-1,1,1)
    else
        data.Dragon.transform.localScale = Vector3(1,1,1)
    end
    if data.attackTimer == 0 and Direction.magnitude <= data.DMRadius then
        data.state = "attackBefore" --进入攻击前摇阶段
        --加入播放攻击前摇音效
        --AM:PlaySFX()
        --在c#类中处理
        data.beforeTimer = data.beforeTime
        CS.UnityEngine.Debug.Log("进入恶龙攻击前摇")
    elseif data.attackTimer == 0 and Direction.magnitude > data.DMRadius then
        DragonFindPlayer(entity)
        data.state = "find" --进入攻击前摇阶段
        data.attackTimer = data.attackCooldown
    end
    return

    end


function DragonAttackBefore(entity, dt)--攻击前摇期间

    local data = entity.dataTable
    if entity.animator then
        entity.animator:SetBool("CanAttack",true)
    end
    -- 攻击逻辑
    data.attackPosition = data.target.transform.position + Vector3(0,1,0) -- 记录将要攻击的位置,y轴向上偏移
    SetDirection(data.attackPosition,data.Dragon) --设置朝向
    data.beforeTimer = math.max(0, data.beforeTimer - dt)--更新时间计时器
    if data.beforeTimer == 0 then --前摇结束开始攻击
        data.state = "attack" --进行状态设置
        entity.animator:SetTrigger("Attack")
        DragonAttack(entity, dt)
    end
end

local roarAudios = {
    "Assets/Audio/Sfx/roar01.wav",
    "Assets/Audio/Sfx/roar02.wav",
}
function DragonAttack(entity)
    --加入播放攻击音效
    local index = math.random(1, #roarAudios)
    AM:PlaySFX(roarAudios[index])

    CS.UnityEngine.Debug.Log("恶龙执行攻击")
    local data = entity.dataTable
    -- 攻击逻辑
    local extraAttack = data.attackPosition - data.Dragon.transform.position
    local attackPosition = data.attackPosition + extraAttack.normalized * data.extraAttackDistance
    --加上额外冲刺距离
    data.Dragon.transform:DOMove(attackPosition, data.attackTime):SetEase(CS.DG.Tweening.Ease.OutCubic):
    OnComplete(function ()
        DragonAttackBack()
    end)

    function DragonAttackBack()   --攻击结束回调函数
        CS.UnityEngine.Debug.Log("调用回调方法")
        if CS.Tool.AnimatorIsInState(entity.animator,"Attack") then --如果在攻击状态下
        entity.animator:SetTrigger("StopAttack")
        data.state = "attackAfter" --进行状态设置
        CS.UnityEngine.Debug.Log("攻击后摇")
        end
        end
end



function DragonAttackAfter(entity, dt)--攻击后摇期间
    --entity.sr.gameObject.transform:DORotate(Vector3(0, 0, 0),0.3)
    CS.UnityEngine.Debug.Log("恶龙攻击后摇")
    local data = entity.dataTable
    data.afterTimer = math.max(0, data.afterTimer - dt)--更新时间计时器
    if data.afterTimer == 0 then
        CS.UnityEngine.Debug.Log("调用飞行方法")
        data.afterTimer = data.afterTime--重置后摇计时器
        data.state = "flyBack"
        entity.animator:SetBool("CanAttack",false)
        entity.animator:SetTrigger("Fly")
        DragonFly(entity)
    end

end


function DragonFly(entity)--飞到指定的点位
    --CS.UnityEngine.Debug.Log()--点位提前放置
    local data = entity.dataTable

    entity.animator:ResetTrigger("StopAttack")--保险起见清楚没有被触发的trigger
    if CS.BossStopPoint.Instance:Equals(nil) then --如果被销毁则返回
        return
    end
    local flyPosition = CS.BossStopPoint.Instance:GetPoint()
    SetDirection(flyPosition,data.Dragon) --设置朝向
    --加入播放煽动翅膀音效
    --AM:PlaySFX()
    data.Dragon.transform:DOMove(flyPosition,data.flyTiming):SetEase(CS.DG.Tweening.Ease.InOutCubic)
    :OnComplete(function ()
    data.state = "idle"
        CS.UnityEngine.Debug.Log("飞回完成")
    data.attackTimer = data.attackCooldown --重置攻击冷却时间计时器
    end)
end

function DragonFindPlayer(entity)--寻找角色
    CS.UnityEngine.Debug.Log("恶龙寻找角色")
    local data = entity.dataTable
    data.attackPosition = data.target.transform.position
    SetDirection( data.attackPosition,data.Dragon) --设置朝向
    local Direction =  data.attackPosition - data.Dragon.transform.position
    if Direction.magnitude > data.DMRadius then
        local point =data.Dragon.transform.position + Direction.normalized*(Direction.magnitude - data.DMRadius+0.5)
        data.Dragon.transform:DOMove( point,data.flyTiming):SetEase(CS.DG.Tweening.Ease.InOutCubic)
              :OnComplete(function ()
            data.state = "idle"
            CS.UnityEngine.Debug.Log("到角色附近")
        end)
    else
        CS.UnityEngine.Debug.Log("已经角色附近")
        data.state = "idle"
    end
end
