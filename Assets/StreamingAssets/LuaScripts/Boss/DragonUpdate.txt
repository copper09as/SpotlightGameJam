local Mathf = CS.UnityEngine.Mathf
function DragonUpdate(entity, dt)
    local data = entity.dataTable
    local target = data.target
    if target == nil then return end

    -- 更新冷却计时
    data.attackTimer = math.max(0, data.attackTimer - dt)
    data.sprintTimer = math.max(0, data.sprintTimer - dt)
    data.flyTime = math.max(0, data.flyTime - dt)

    -- 根据状态执行行为
    if data.state == "idle" then --待机不执行动作
        DragonIdle(entity, dt)
    elseif data.state == "attackBefore" then
        DragonAttackBefore(entity, dt)
    elseif data.state == "attack" then
        DragonAttack(entity, dt)
    elseif data.state == "fly" then
        DragonFly(entity, dt)
    --elseif data.state == "sprint" then
    --    BossSprint(entity, dt)
    end
end


function DragonIdle(entity, dt) --什么都不做
    return
end


function DragonAttackBefore(entity, dt)--攻击前摇期间

    local data = entity.dataTable
    if entity.animator then
        entity.animator:SetBool("CanAttack",true)
    end
    -- 攻击逻辑
    data.attackPosition = data.target.transform.position -- 记录将要攻击的位置
    local normalizedDirection = data.attackPosition.normalized
    local distance =  data.attackPosition.magnitude
    local angle =
    Mathf.Atan2(normalizedDirection.y,normalizedDirection.x) * Mathf.Rad2Deg + 90;--巨龙朝向

    entity.transform.rotation = CS.UnityEngine.Euler(0,0,angle)


    data.beforeTimer = math.max(0, data.beforeTimer - dt)--更新时间计时器
    if data.beforeTimer == 0 then --前摇结束开始攻击
        entity.animator:SetTrigger("Attack")
        data.state = "attack" --进行状态设置
    end

end

function DragonAttack(entity, dt)

    local data = entity.dataTable
    -- 攻击逻辑
    entity.transform:DOMove(data.attackPosition,3):SetEase(CS.Ease.OutCubic):
    OnComplete(function ()
    entity.animator:SetTrigger("Attack")
    data.state = "attack" --进行状态设置
    end)


    data.attackTimer = math.max(0, data.attackTimer - dt)
    function DragonAttackBack()   --攻击结束回调函数
        entity.animator:SetTrigger("StopAttack")
        data.state = "" --进行状态设置
    end

end
