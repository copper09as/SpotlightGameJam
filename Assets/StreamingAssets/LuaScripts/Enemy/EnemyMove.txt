local DM = CS.DetectionManager--检测管理器
local Vector2 = CS.UnityEngine.Vector2
local Tool = CS.Tool

local function EnemySeeLeftAndRight(enemyEntity,distance)
    local origin = Tool.ToVector2(enemyEntity.transform.position)
    local length = Tool.GetColliderWidthLua(enemyEntity.col)/2 + distance

    local isLeftEdge,LeftHit= DM.Raycast2DNoLayerOutHit(origin-Vector2(Tool.GetColliderWidthLua(enemyEntity.col)/2+0.01,0),
            Vector2(-1,0),length)
    local isRightEdge,RightHit = DM.Raycast2DNoLayerOutHit(origin+Vector2(Tool.GetColliderWidthLua(enemyEntity.col)/2+0.01,0),
            Vector2(1,0),length)

    local answerA,answerB = false
    if isLeftEdge then
        local EntityA = LeftHit.collider:GetComponent("Entity")
        if EntityA ~= nil then
            answerA = EntityA.dataTable.enemyNeedTurnRound or false

        end
    end
    if isRightEdge then
        local EntityB = RightHit.collider:GetComponent("Entity")
        if EntityB ~= nil then
            answerB = EntityB.dataTable.enemyNeedTurnRound or false
        end
    end
    return answerA,answerB
end

function EnemyMove(entity)
    local data = entity.dataTable
    if entity.rb == nil then
        return
    end

    local rb = entity.rb
    local EnemySpeed = data.EnemySpeed

    local velocity = rb.velocity

    -- 边缘检测
    local x = DM.MultiRayGroundCheck(entity.col, 3,0.2, "Ground")
    local leftHit, rightHit  = EnemySeeLeftAndRight(entity,0.1)
    local isLeftEdge = x.isLeftEdge or leftHit-- 是否在左边缘
    local isRightEdge = x.isRightEdge or rightHit-- 是否在右边缘

    -- 停顿处理
    if data.isPausing then
        data.pauseTimer = data.pauseTimer + CS.UnityEngine.Time.deltaTime
        velocity.x = 0
        rb.velocity = velocity
        if data.pauseTimer >= data.pauseDuration then
            data.isPausing = false
            data.pauseTimer = 0
            -- 改变方向
            data.moveDirection = not data.moveDirection
            entity.sr.flipX = not entity.sr.flipX

        end
        return
    end


    -- 检测到边缘时开始停顿
    if (isLeftEdge and not data.moveDirection ) or (isRightEdge and data.moveDirection ) then
        velocity.x = 0
        rb.velocity = velocity
        data.isPausing = true
        return
    end

    -- 正常移动
    velocity.x = data.moveDirection and EnemySpeed or -EnemySpeed
    rb.velocity = velocity

    if isLeftEdge then
        print("在左边缘")
    end

end
