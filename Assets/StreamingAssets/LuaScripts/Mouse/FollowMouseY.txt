local GC = CS.GameController
local Screen = CS.UnityEngine.Screen
local Mathf = CS.UnityEngine.Mathf
local Vector3 = CS.UnityEngine.Vector3
local Camera = CS.UnityEngine.Camera
local Time = CS.UnityEngine.Time


function FollowMouseY(entity)
    if entity == nil or entity.transform == nil then return end

    local cam = Camera.main
    local data = entity.dataTable
    local mousePos = GC.GetScreenPos()
    local dt = Time.unscaledDeltaTime

    -- 🧮 屏幕比例
    local scaleX = Screen.width / 1920
    local scaleY = Screen.height / 1080

    -- 🧭 根据当前屏幕重新计算边界
    local limitYMin = (data.baseLimitYMin or 150) * scaleY
    local limitYMax = (data.baseLimitYMax or 1000) * scaleY

    -- 限制鼠标Y范围
    mousePos.y = Mathf.Clamp(mousePos.y, limitYMin, limitYMax)

    -- 获取当前屏幕坐标
    local screenPos = cam:WorldToScreenPoint(entity.transform.position)

    -- 匀速移动
    local targetY = mousePos.y
    local moveSpeed = (data.scrollSpeed or 1000) * scaleY
    local diffY = targetY - screenPos.y

    if math.abs(diffY) > 0.1 then
        local moveStep = moveSpeed * dt
        if math.abs(diffY) < moveStep then
            screenPos.y = targetY
        else
            screenPos.y = screenPos.y + moveStep * (diffY > 0 and 1 or -1)
        end
    end

    screenPos.y = Mathf.Clamp(screenPos.y, limitYMin, limitYMax)

    local worldPos = cam:ScreenToWorldPoint(Vector3(screenPos.x, screenPos.y, screenPos.z))
    entity.transform.position = Vector3(worldPos.x, worldPos.y, entity.transform.position.z)

    local rangeY = limitYMax - limitYMin
    data.scrollProgress = (screenPos.y - limitYMin) / rangeY
end
