local GC = CS.GameController

function FollowMouseX(entity)
    if entity == nil or entity.transform == nil then return end

    local cam = CS.UnityEngine.Camera.main
    local data = entity.dataTable
    local mousePos = GC.GetScreenPos()
    local dt = CS.UnityEngine.Time.unscaledDeltaTime

    -- 限制鼠标在屏幕X范围内
    mousePos.x = CS.UnityEngine.Mathf.Clamp(mousePos.x, data.limitXMin, data.limitXMax)

    -- 获取当前滑块的屏幕坐标
    local screenPos = cam:WorldToScreenPoint(entity.transform.position)

    -- 匀速移动
    local targetX = mousePos.x
    local moveSpeed = data.scrollSpeedX or 1000  -- 每秒移动的像素速度
    local diffX = targetX - screenPos.x

    if math.abs(diffX) > 0.1 then
        local moveStep = moveSpeed * dt
        if math.abs(diffX) < moveStep then
            screenPos.x = targetX
        else
            screenPos.x = screenPos.x + moveStep * (diffX > 0 and 1 or -1)
        end
    end

    -- 限制最终屏幕位置
    screenPos.x = CS.UnityEngine.Mathf.Clamp(screenPos.x, data.limitXMin, data.limitXMax)

    -- 转回世界坐标并应用
    local worldPos = cam:ScreenToWorldPoint(CS.UnityEngine.Vector3(screenPos.x, screenPos.y, screenPos.z))
    entity.transform.position = CS.UnityEngine.Vector3(worldPos.x, entity.transform.position.y, entity.transform.position.z)

    -- 直接计算 scrollProgress
    local rangeX = data.limitXMax - data.limitXMin
    data.scrollProgress = (screenPos.x - data.limitXMin) / rangeX
end
