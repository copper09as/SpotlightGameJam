--碰撞到角色时候触发钥匙跟随角色,当角色与钥匙发生碰撞的时候就进行环绕

function KeyStartFollow(entity,character)
    CS.UnityEngine.Debug.Log(character.dataId)
    if character.dataId ~= 20000 and character.dataId ~= 20022then
        return
    end
    entity.transform.parent = nil --父物体置空
    CS.EntityUIManager.Instance:RemoveObject(entity.gameObject)
    local data  = entity.dataTable or {}
    --存放在luaTable中的数据
    data.player = character--存放要跟随的角色实体
    data.lastPosition = character.transform.position --存放角色上一帧的位置

    data.canFollow = true
    entity.col.enabled = false
    data.tweener = entity.transform:DOMove(character.transform.position,
            data.followDuration):SetAutoKill(false)
    --CS.UnityEngine.Debug.Log("碰到角色")
end
